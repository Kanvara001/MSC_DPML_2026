Masked image modeling: ทาย visual token ของ patch ที่ถูกปิด

models/lib/BEiT_v2/modeling_pretrain.py · VisionTransformerForMaskedImageModeling 
def forward_features(self, x, bool_masked_pos):
    x = self.patch_embed(x)                       # (B,196,768)
    mask_token = self.mask_token.expand(B, 196, -1)
    # replace the masked visual tokens by mask_token
    w = bool_masked_pos.unsqueeze(-1).type_as(mask_token)
    x = x * (1 - w) + mask_token * w
    x = torch.cat((cls_tokens, x), dim=1)        # (B,197,768)
    for blk in self.blocks:                     # 12 block
        x = blk(x, rel_pos_bias=rel_pos_bias)
    return self.norm(x)

def forward(self, x, bool_masked_pos):
    x = self.forward_features(x, bool_masked_pos)[:, 1:]
    return self.lm_head(x[bool_masked_pos])     # Linear(768 → 8192)

  1. แบ่งภาพใบหน้าจริงเป็น 14 × 14 = 196 patch

def forward(self, x):
    B, T, C, H, W = x.size()
    x = x.flatten(0,1)
    clstoken  = self.backbone_model(x)          # f_M  main
    clstoken2 = self.backbone_model2(x)         # f_A  auxiliary
    pearson_loss = pearson_correlation_loss(clstoken.detach(), clstoken2)
    pred1 = self.header(self.norm_for_cls(clstoken))
    pred2 = self.header(self.norm_for_cls(clstoken2))
    pred1_un = uncertainty(pred1)
    pred2_un = uncertainty(pred2)
    weight = (-10)*torch.cat([pred1_un,pred2_un], dim=-1)
    w = torch.nn.functional.softmax(weight, dim=-1)
    weighted_output = torch.matmul(w.unsqueeze(1),
        torch.stack((clstoken, clstoken2), dim=1)).squeeze(1)
    pred = self.header(self.norm_for_cls(weighted_output))
    output  = pred.view(B, T, -1)               # fused
    output1 = pred1.view(B, T, -1)              # main
    output2 = pred2.view(B, T, -1)              # auxiliary
    return output, output1, output2, pearson_loss

Ldec คือค่า Pearson โดยตรง การ minimize จึงดันให้ 2 branch มี correlation ติดลบ

models/custom.py · pearson_correlation_loss (55–68) 
def pearson_correlation_loss(x, y):
    mean_x = torch.mean(x, dim=-1, keepdim=True)  # เฉลี่ยตามมิติ feature
    mean_y = torch.mean(y, dim=-1, keepdim=True)
    xm = x.sub(mean_x);  ym = y.sub(mean_y)
    Conv_xy = torch.bmm(xm.unsqueeze(1),
                        ym.unsqueeze(2)).squeeze()/x.size(-1)
    std_x = torch.std(x, dim=-1);  std_y = torch.std(y, dim=-1)
    pearson_cor = Conv_xy/(std_x*std_y)            # 1 ค่าต่อ 1 ภาพ
    r = torch.clamp(pearson_cor.mean(), -1.0, 1.0)   # เฉลี่ยทั้ง batch
    return r

S = torch.sum(alpha, dim=1, keepdim=True)
A = torch.sum(y * (torch.log(S) - torch.log(alpha)), dim=1)    # L_EDL

def utom_adjust(p, u, lam=0.02, eps=1e-7):
    return np.tanh(lam * p / (2 * (u + eps)))        # p′ (Sec. III-D)

def best_threshold(y, s):                          # Eq. 10–11
    o = np.argsort(s)                      # ทุกค่า score เป็นผู้สมัคร τ
    real_below = cumsum(real[o])           # real ที่ s < τ (ถูก)
    fake_above = n_fake − cumsum(fake[o])  # fake ที่ s ≥ τ (ถูก)
    return s[o][argmax(real_below + fake_above)]
ขั้นตอนตอนใช้งานจริง
  1. หา τUTOM บน source ที่มี label (เช่น FF++)
  2. วิดีโอใหม่: เฉลี่ย p และ u ของ 8 frame
  3. คำนวณ p′ แล้วตัดสินด้วย τUTOM โดยไม่ต้องมี label ของ target
